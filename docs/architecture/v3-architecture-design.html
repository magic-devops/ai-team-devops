# apis/openapi/memory.yaml (部分)
paths:
  /api/v1/memory/{tenant_code}/{project_code}:
    post:
      summary: 写入记忆
      requestBody:
        content:
          application/json:
            schema:
              type: object
              properties:
                content:
                  type: string
                  description: 记忆内容
                metadata:
                  type: object
                  description: 元数据
                memory_type:
                  type: string
                  enum: [short_term, long_term, episodic]
              required: [content, memory_type]
      responses:
        '200':
          description: 写入成功
          content:
            application/json:
              schema:
                type: object
                properties:
                  memory_id:
                    type: string
                    format: uuid

数据库 Schema 设计

-- 租户表
CREATE TABLE tenants (
    tenant_code VARCHAR(32) PRIMARY KEY,
    tenant_name VARCHAR(128) NOT NULL,
    status TINYINT UNSIGNED NOT NULL DEFAULT 1,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP 
                ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='租户表';

-- 业务表
CREATE TABLE businesses (
    business_code VARCHAR(32) PRIMARY KEY,
    tenant_code VARCHAR(32) NOT NULL,
    business_name VARCHAR(128) NOT NULL,
    status TINYINT UNSIGNED NOT NULL DEFAULT 1,
    INDEX idx_tenant_code (tenant_code),
    FOREIGN KEY (tenant_code) REFERENCES tenants(tenant_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='业务表';

-- 项目表
CREATE TABLE projects (
    project_code VARCHAR(32) PRIMARY KEY,
    business_code VARCHAR(32) NOT NULL,
    tenant_code VARCHAR(32) NOT NULL,
    project_name VARCHAR(128) NOT NULL,
    status TINYINT UNSIGNED NOT NULL DEFAULT 1,
    INDEX idx_tenant_business (tenant_code, business_code),
    FOREIGN KEY (business_code) REFERENCES businesses(business_code),
    FOREIGN KEY (tenant_code) REFERENCES tenants(tenant_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='项目表';
